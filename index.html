트위터 RT 이벤트 추첨기

리트윗 참여자 목록을 넣고, 공정하게 당첨자를 뽑으세요.

1. 참여자 입력

Twitter/X RT 목록을 콘솔에서 자동 추출하는 스크립트

RT한 사람 목록 화면을 열고 F12 또는 Ctrl+Shift+I로 개발자 도구를 연 뒤, Console 탭에 아래 코드를 붙여넣으세요. 완료되면 사용자명 목록이 클립보드에 복사됩니다.

(async () => {
  const users = new Set();
  const ignored = new Set(["home", "explore", "notifications", "messages", "i", "search", "settings"]);
  let lastHeight = 0, stable = 0;
  console.log("RT 사용자명 수집 시작");
  while (stable < 5) {
    document.querySelectorAll('a[href^="/"]').forEach((a) => {
      const match = a.getAttribute("href").match(/^\/([A-Za-z0-9_]{1,15})$/);
      if (match && !ignored.has(match[1])) users.add(match[1]);
    });
    window.scrollBy(0, 900);
    await new Promise((resolve) => setTimeout(resolve, 850));
    const height = document.documentElement.scrollHeight;
    if (height === lastHeight) stable += 1;
    else { stable = 0; lastHeight = height; }
    console.log(`현재 ${users.size}명 수집`);
  }
  const list = [...users].join("\n");
  await navigator.clipboard.writeText(list);
  console.log(`완료: ${users.size}명을 클립보드에 복사했습니다.`);
})();
